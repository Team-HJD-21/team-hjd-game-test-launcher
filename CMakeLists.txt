cmake_minimum_required(VERSION 3.24)
project(TeamHJDGameTestLauncher VERSION 0.1.3 LANGUAGES C CXX)
if(WIN32)
  enable_language(RC)
elseif(NOT APPLE)
  message(FATAL_ERROR "The launcher supports Windows and macOS.")
endif()
set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_AUTOMOC ON)
if(MSVC)
  add_compile_options(/utf-8 /W4 /permissive- /EHsc)
  if(HJD_MSVC_INCLUDE_PREFIX)
    set(CMAKE_CL_SHOWINCLUDES_PREFIX "${HJD_MSVC_INCLUDE_PREFIX}")
  endif()
endif()
find_package(Qt6 6.8 REQUIRED COMPONENTS Core Gui Quick QuickControls2 Network Concurrent)
qt_standard_project_setup(REQUIRES 6.8)
if(WIN32)
  configure_file(src/app.rc.in "${CMAKE_CURRENT_BINARY_DIR}/brand.rc" @ONLY)
endif()
add_library(miniz STATIC vendor/miniz/miniz.c vendor/miniz/miniz_tdef.c vendor/miniz/miniz_tinfl.c vendor/miniz/miniz_zip.c)
set_target_properties(miniz PROPERTIES AUTOMOC OFF)
target_include_directories(miniz PUBLIC vendor/miniz)
target_compile_definitions(miniz PUBLIC MINIZ_NO_ZLIB_COMPATIBLE_NAMES)
add_library(launcher_core STATIC src/core/Core.cpp src/core/Http.cpp src/core/Store.cpp src/core/CloudflareLogin.cpp)
target_include_directories(launcher_core PUBLIC src)
target_link_libraries(launcher_core PUBLIC Qt6::Core Qt6::Network miniz)
if(WIN32)
  target_link_libraries(launcher_core PUBLIC bcrypt crypt32 advapi32)
else()
  target_link_libraries(launcher_core PUBLIC "-framework Security" "-framework CoreFoundation" "-framework ApplicationServices")
endif()
target_compile_definitions(launcher_core PUBLIC NOMINMAX WIN32_LEAN_AND_MEAN HJD_VERSION="${PROJECT_VERSION}")
qt_add_executable(TeamHJDClient WIN32 src/client/main.cpp src/client/Controller.cpp src/client/Controller.h src/client/WindowFrame.cpp)
target_link_libraries(TeamHJDClient PRIVATE launcher_core Qt6::Gui Qt6::Quick Qt6::QuickControls2 Qt6::Concurrent)
qt_add_resources(TeamHJDClient "ui" PREFIX "/" FILES qml/Main.qml qml/ActionButton.qml qml/AppMenu.qml qml/AppMenuItem.qml qml/AppMenuSeparator.qml qml/AppToolTip.qml qml/AppCheckBox.qml assets/brand/TeamHJD-Logo.png assets/brand/TeamHJD-Simple-Logo.png assets/icons/bell.svg assets/games/hose-for-hire-logo.png)
qt_add_executable(TeamHJDLauncher WIN32 src/bootstrap/main.cpp)
target_link_libraries(TeamHJDLauncher PRIVATE launcher_core)
if(WIN32)
target_sources(TeamHJDClient PRIVATE "${CMAKE_CURRENT_BINARY_DIR}/brand.rc")
target_sources(TeamHJDLauncher PRIVATE "${CMAKE_CURRENT_BINARY_DIR}/brand.rc")
target_link_libraries(TeamHJDClient PRIVATE user32 dwmapi)
target_link_libraries(TeamHJDLauncher PRIVATE user32)
else()
  set_source_files_properties(assets/brand/TeamHJD.icns PROPERTIES MACOSX_PACKAGE_LOCATION Resources)
  foreach(app TeamHJDClient TeamHJDLauncher)
    target_sources(${app} PRIVATE assets/brand/TeamHJD.icns)
    set_target_properties(${app} PROPERTIES MACOSX_BUNDLE TRUE
      MACOSX_BUNDLE_GUI_IDENTIFIER "com.teamhjd.${app}"
      MACOSX_BUNDLE_BUNDLE_NAME "TeamHJD Game Test Launcher"
      MACOSX_BUNDLE_BUNDLE_VERSION "${PROJECT_VERSION}"
      MACOSX_BUNDLE_SHORT_VERSION_STRING "${PROJECT_VERSION}"
      MACOSX_BUNDLE_ICON_FILE TeamHJD.icns)
  endforeach()
endif()
enable_testing()
if(WIN32)
add_executable(LauncherCoreTests tests/CoreTests.cpp)
target_link_libraries(LauncherCoreTests PRIVATE launcher_core)
add_executable(LauncherFixtureGame WIN32 tests/FixtureGame.cpp)
target_link_libraries(LauncherFixtureGame PRIVATE user32 shell32)
add_executable(LauncherWindowFrameTests tests/WindowFrameTests.cpp src/client/WindowFrame.cpp)
target_include_directories(LauncherWindowFrameTests PRIVATE src)
target_compile_definitions(LauncherWindowFrameTests PRIVATE NOMINMAX WIN32_LEAN_AND_MEAN)
target_link_libraries(LauncherWindowFrameTests PRIVATE Qt6::Gui Qt6::Quick user32 dwmapi)
add_executable(LauncherControllerUpdateTests tests/ControllerUpdateTests.cpp src/client/Controller.cpp src/client/Controller.h)
target_link_libraries(LauncherControllerUpdateTests PRIVATE launcher_core Qt6::Gui Qt6::Concurrent user32)
add_test(NAME core COMMAND LauncherCoreTests)
set_tests_properties(core PROPERTIES TIMEOUT 90)
add_test(NAME gui_startup_shutdown COMMAND TeamHJDClient --data-root "${CMAKE_CURRENT_BINARY_DIR}/../.local/gui-ctest" --quit-after-ms 700)
set_tests_properties(gui_startup_shutdown PROPERTIES TIMEOUT 20)
add_test(NAME gui_window_frame COMMAND LauncherWindowFrameTests)
set_tests_properties(gui_window_frame PROPERTIES TIMEOUT 12)
find_program(HJD_PWSH_EXECUTABLE NAMES pwsh REQUIRED)
get_filename_component(HJD_TEST_QT_ROOT "${Qt6_DIR}/../../.." ABSOLUTE)
add_test(NAME controller_manual_update COMMAND "${HJD_PWSH_EXECUTABLE}" -NoProfile -File "${CMAKE_CURRENT_SOURCE_DIR}/tools/verify-manual-update.ps1" -QtRoot "${HJD_TEST_QT_ROOT}" -Harness "$<TARGET_FILE:LauncherControllerUpdateTests>" -Probe "$<TARGET_FILE:LauncherCoreTests>" -Fixture "$<TARGET_FILE:LauncherFixtureGame>")
set_tests_properties(controller_manual_update PROPERTIES TIMEOUT 120)
else()
  add_executable(LauncherCoreTests tests/MacCoreTests.cpp)
  target_link_libraries(LauncherCoreTests PRIVATE launcher_core)
  add_executable(LauncherFixtureGame tests/MacFixtureGame.cpp)
  add_test(NAME core COMMAND LauncherCoreTests)
  set_tests_properties(core PROPERTIES TIMEOUT 90)
  add_test(NAME gui_startup_shutdown COMMAND TeamHJDClient --data-root "${CMAKE_CURRENT_BINARY_DIR}/../.local/gui-ctest" --quit-after-ms 700)
  set_tests_properties(gui_startup_shutdown PROPERTIES TIMEOUT 20)
endif()
